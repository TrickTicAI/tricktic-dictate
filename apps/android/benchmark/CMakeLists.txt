cmake_minimum_required(VERSION 3.22)
project(writer_android_bench LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# A single static CPU backend: no dynamic plugin discovery or remote RPC backend.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
foreach(backend CUDA METAL VULKAN OPENCL HEXAGON RPC BLAS CANN SYCL MUSA HIP OPENVINO VIRTGPU ET ZDNN ZENDNN)
  set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(writer-android-bench main.cpp)
target_include_directories(writer-android-bench PRIVATE "${WHISPER_SOURCE}/examples")
target_link_libraries(writer-android-bench PRIVATE whisper)
target_compile_options(writer-android-bench PRIVATE -Wall -Wextra -Werror)
